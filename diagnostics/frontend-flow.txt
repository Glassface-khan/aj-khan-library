  handleAltCoverFiles(i, e) {
    const files = e.target.files ? Array.from(e.target.files) : [];
    if (!files.length) return;
    const readOne = (file) => new Promise((resolve, reject) => {
      const img = new Image();
      const reader = new FileReader();
      reader.onload = (ev) => {
        img.onload = () => {
          const maxW = 480;
          const scale = Math.min(1, maxW / img.width);
          const canvas = document.createElement('canvas');
          canvas.width = img.width * scale;
          canvas.height = img.height * scale;
          const ctx = canvas.getContext('2d');
          ctx.drawImage(img, 0, 0, canvas.width, canvas.height);
          let dataUrl = canvas.toDataURL('image/jpeg', 0.65);
          if (dataUrl.length > 60000) dataUrl = canvas.toDataURL('image/jpeg', 0.4);
          if (dataUrl.length > 60000) { reject('zu groß'); return; }
          resolve(dataUrl);
        };
        img.onerror = () => reject('kein Bild');
        img.src = ev.target.result;
      };
      reader.onerror = () => reject('Lesefehler');
      reader.readAsDataURL(file);
    });
    Promise.all(files.map((f) => readOne(f).catch(() => null))).then((results) => {
      const okOnes = results.filter(Boolean);
      const skipped = results.length - okOnes.length;
      if (okOnes.length) {
        this.setState((s) => {
          const books = s.books.slice();
          const existing = books[i].altCoversInline || [];
          books[i] = { ...books[i], altCoversInline: existing.concat(okOnes) };
          return { books };
        });
      }
      if (skipped) window.alert(skipped + ' Bild(er) konnten nicht hinzugefügt werden (zu groß oder kein gültiges Bild). Bitte kleinere/einfachere Bilder versuchen.');
    });
  }

      const existing = (books[i].altCoversInline || []).slice();
      existing.splice(idx, 1);
      books[i] = { ...books[i], altCoversInline: existing };
      return { books };
    });
  }
